cmake_minimum_required(VERSION 3.20)
project(package_manager_consumers LANGUAGES C CXX)
enable_testing()
find_package(RobotweaxSRT 0.2.5 EXACT CONFIG REQUIRED)
add_executable(robotweax_native ../../tests/package_consumer/native_c_consumer.c)
set_target_properties(robotweax_native PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(robotweax_native PRIVATE RobotweaxSRT::srt)
add_test(NAME robotweax_native COMMAND robotweax_native)
add_executable(robotweax_cpp ../../tests/package_consumer/cpp_consumer.cpp)
target_compile_features(robotweax_cpp PRIVATE cxx_std_20)
target_link_libraries(robotweax_cpp PRIVATE RobotweaxSRT::srt)
add_test(NAME robotweax_cpp COMMAND robotweax_cpp)
option(TEST_HAIVISION "Test filesystem coexistence with Haivision" ON)
if(TEST_HAIVISION)
  # Separate executables: filesystem coexistence is not in-process coexistence.
  find_path(HAIVISION_INCLUDE srt/srt.h PATHS "${HAIVISION_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(HAIVISION_LIBRARY NAMES srt PATHS "${HAIVISION_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  find_library(HAIVISION_DEBUG_LIBRARY NAMES srt PATHS "${HAIVISION_PREFIX}/debug/lib" NO_DEFAULT_PATH)
  if(NOT HAIVISION_DEBUG_LIBRARY)
    set(HAIVISION_DEBUG_LIBRARY "${HAIVISION_LIBRARY}")
  endif()
  add_library(haivision_package UNKNOWN IMPORTED)
  set_target_properties(haivision_package PROPERTIES
    IMPORTED_LOCATION "${HAIVISION_LIBRARY}"
    IMPORTED_LOCATION_DEBUG "${HAIVISION_DEBUG_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${HAIVISION_INCLUDE}")
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(haivision haivision.cpp)
  target_link_libraries(haivision PRIVATE haivision_package OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
  if(WIN32)
    target_link_libraries(haivision PRIVATE ws2_32 crypt32)
  endif()
  add_test(NAME haivision COMMAND haivision)
endif()
