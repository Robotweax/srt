cmake_minimum_required(VERSION 3.20)
project(peer_authentication_peer LANGUAGES C CXX)
if(DEFINED HAIVISION_SOURCE)
    set(ENABLE_APPS OFF CACHE BOOL "" FORCE)
    set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
    set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
    set(ENABLE_ENCRYPTION ON CACHE BOOL "" FORCE)
    set(ENABLE_AEAD_API_PREVIEW ON CACHE BOOL "" FORCE)
    # The reference exports bare ssl/crypto names on some source builds.
    find_package(OpenSSL REQUIRED)
    get_filename_component(PEER_OPENSSL_LIBRARY_DIR "${OPENSSL_CRYPTO_LIBRARY}" DIRECTORY)
    link_directories("${PEER_OPENSSL_LIBRARY_DIR}")
    add_subdirectory("${HAIVISION_SOURCE}" reference)
    set(PEER_LIBRARY srt_static)
else()
    find_package(RobotweaxSRT CONFIG REQUIRED)
    set(PEER_LIBRARY RobotweaxSRT::srt)
endif()
add_executable(peer_authentication_peer peer.cpp)
target_compile_features(peer_authentication_peer PRIVATE cxx_std_20)
target_link_libraries(peer_authentication_peer PRIVATE ${PEER_LIBRARY})
if(DEFINED HAIVISION_SOURCE)
    target_compile_definitions(peer_authentication_peer PRIVATE PEERAUTH_REFERENCE ENABLE_AEAD_API_PREVIEW)
    target_include_directories(peer_authentication_peer PRIVATE "${HAIVISION_SOURCE}/srtcore" "${CMAKE_CURRENT_BINARY_DIR}/reference")
endif()
if(WIN32)
    target_link_libraries(peer_authentication_peer PRIVATE ws2_32)
endif()

if(DEFINED ROBOTWEAX_NATIVE_SOURCE AND NOT DEFINED HAIVISION_SOURCE)
    add_executable(peer_authentication_wire wire.cpp)
    target_compile_features(peer_authentication_wire PRIVATE cxx_std_20)
    target_include_directories(peer_authentication_wire PRIVATE "${ROBOTWEAX_NATIVE_SOURCE}/include")
    target_link_libraries(peer_authentication_wire PRIVATE RobotweaxSRT::srt)
endif()
