cmake_minimum_required(VERSION 3.20)
project(mbedtls_qualification LANGUAGES CXX)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
list(APPEND CMAKE_MODULE_PATH "${ROOT}/cmake")
find_package(MbedTLS 3.6.7 REQUIRED MODULE)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
foreach(backend IN ITEMS mbedtls openssl)
    add_library(${backend}_oracle OBJECT "${ROOT}/src/crypto_${backend}.cpp")
    target_compile_features(${backend}_oracle PRIVATE cxx_std_20)
    target_include_directories(${backend}_oracle PRIVATE "${ROOT}/include")
    target_compile_definitions(${backend}_oracle PRIVATE
        default_crypto_provider=${backend}_test_provider)
endforeach()
target_link_libraries(mbedtls_oracle PRIVATE MbedTLS::mbedcrypto)
target_link_libraries(openssl_oracle PRIVATE OpenSSL::Crypto)
add_executable(compare_mbedtls main.cpp
    $<TARGET_OBJECTS:mbedtls_oracle> $<TARGET_OBJECTS:openssl_oracle>)
target_compile_features(compare_mbedtls PRIVATE cxx_std_20)
target_include_directories(compare_mbedtls PRIVATE "${ROOT}/include")
target_link_libraries(compare_mbedtls PRIVATE MbedTLS::mbedcrypto OpenSSL::Crypto Threads::Threads)
enable_testing()
add_test(NAME mbedtls_openssl_differential COMMAND compare_mbedtls)
