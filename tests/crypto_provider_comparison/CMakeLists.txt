cmake_minimum_required(VERSION 3.20)
project(crypto_provider_comparison LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "BCrypt comparison requires Windows")
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_library(openssl_oracle OBJECT "${ROOT}/src/crypto_openssl.cpp")
target_compile_features(openssl_oracle PRIVATE cxx_std_20)
target_include_directories(openssl_oracle PRIVATE "${ROOT}/include")
target_compile_definitions(openssl_oracle PRIVATE default_crypto_provider=openssl_test_provider)
target_link_libraries(openssl_oracle PRIVATE OpenSSL::Crypto)
add_executable(compare_providers main.cpp "${ROOT}/src/crypto_bcrypt.cpp" $<TARGET_OBJECTS:openssl_oracle>)
target_compile_features(compare_providers PRIVATE cxx_std_20)
target_include_directories(compare_providers PRIVATE "${ROOT}/include")
target_compile_definitions(compare_providers PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(compare_providers PRIVATE OpenSSL::Crypto bcrypt)
enable_testing()
add_test(NAME compare_providers COMMAND compare_providers)
add_executable(provider_failures failures.cpp)
target_compile_features(provider_failures PRIVATE cxx_std_20)
target_include_directories(provider_failures PRIVATE "${ROOT}/include")
target_compile_definitions(provider_failures PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(provider_failures PRIVATE bcrypt)
add_test(NAME provider_failures COMMAND provider_failures)
