foreach(application IN ITEMS contribution recovery device multicast)
    add_executable(robotweax_srt_${application} ${application}.cpp)
    target_link_libraries(robotweax_srt_${application} PRIVATE RobotweaxSRT::srt)
    target_compile_features(robotweax_srt_${application} PRIVATE cxx_std_20)
    set_target_properties(robotweax_srt_${application} PROPERTIES CXX_EXTENSIONS OFF)
    if(WIN32)
        target_link_libraries(robotweax_srt_${application} PRIVATE ws2_32)
    endif()
endforeach()

    target_include_directories(robotweax_srt_multicast PRIVATE ${PROJECT_SOURCE_DIR}/src)
    if(ROBOTWEAX_SRT_BUILD_TESTS)

        target_compile_definitions(robotweax_srt_multicast_test PRIVATE ROBOTWEAX_SAMPLE_TESTING ROBOTWEAX_SAMPLE_HYBRID)
        target_include_directories(robotweax_srt_multicast_test PRIVATE ${PROJECT_SOURCE_DIR}/src)
        if(WIN32)
            target_link_libraries(robotweax_srt_multicast_test PRIVATE ws2_32)
        endif()
    endif()
endif()
